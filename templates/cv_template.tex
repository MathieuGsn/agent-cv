\newcommand{\cvsectionrule}{\titlerule[0.6pt]}

  {\large\bfseries}
  {}
  {0pt}
  {}
  [\vspace{2pt}\cvsectionrule]

% partopsep=0pt est essentiel : sans ca, enumitem ajoute un espace
% supplementaire quand la liste suit un saut de ligne force (\\), ce qui
% cree des "lignes vides" parasites entre un titre d'entree et ses puces.
\newlist{cvitems}{itemize}{2}
\setlist[cvitems]{leftmargin=1.1em, itemsep=0.5pt, parsep=0pt, topsep=1.5pt, partopsep=0pt, label=--}
